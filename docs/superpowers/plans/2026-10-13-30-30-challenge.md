# 30/30 Challenge Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A single-player 30/30 Challenge next to the leagues: pick Over/Under on all 30 teams, save and edit until a
schedule-derived global deadline, lock every entry against one shared frozen FanDuel snapshot, track projected and final
scores, and compare on a public leaderboard.

**Architecture:** Pure rules in `src/lib/challenge/` (parsing, reference-line rebasing, movement, grid filters, opening
game, per-team freeze, phase, scoring and ranking — scoring is `evaluateCall`). Four new tables (`challenge_seasons`,
`challenge_line_snapshots`, `challenge_entries`, `challenge_picks`) behind `src/db/challenge.ts`; every write is one
transaction that locks the season row. `src/server/challenge.ts` prepares each request (schedule check, FanDuel
snapshot, freeze when due). One JSON endpoint (`PUT /api/challenge/entry`), one daily cron, and five pages under
`/challenge` with their own shell.

**Tech Stack:** Next.js 16.4 App Router (Cache Components), React 19, TypeScript strict, Tailwind CSS v4, Drizzle ORM
0.45 (Postgres / PGlite), Vitest 4.1, lucide-react. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-10-13-30-30-challenge-design.md`. Read `CLAUDE.md` first: it is the project's
rulebook (tokens, formatting, Suspense pattern, responsive rules).

## Global Constraints

- Scoring only through `evaluateCall` (`src/lib/scoring.ts`): +1/−1 base, ±0.1 per win of margin, push 0. No fades, no awards.
- Zero games → "Not available", never 0 points. Unsettled final → "Pending". Missing frozen line → "Unavailable", excluded for everyone.
- `deadline = openingTipoff − CHALLENGE.lockMinutesBeforeTipoff (30)` minutes; never a hardcoded date.
- Writes at or after the deadline are refused by the server using the database clock read after the season-row lock.
- Frozen lines only from snapshots with `observed_at < deadline`; never fabricated, never zero.
- League code and `leagues.lines` are never touched by challenge code.
- Dark theme tokens only (`bg-ink-850`, `text-fog-400`, `text-accent`, `bg-under`, …); Over = lime (`over`), Under = purple (`under`).
- Numbers through `src/lib/format.ts` (one decimal for lines; two decimals for points, U+2212 minus).
- Wrap anything that reads cookies or the database in `<Suspense>`; `getDb()` already awaits `connection()`.
- No horizontal page scroll at 375 px and 1440 px.
- Every commit message ends with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

## Review Focus

- A stale tab saving after the deadline: the server must refuse (`challenge_locked`) even if the page still shows buttons — Task 3 test "refuses a save at and after the deadline".
- Saving unchanged picks after lines moved must keep the old reference line and time — Task 1 and Task 3 tests.
- A FanDuel read that happens after the deadline must never feed the freeze, even if recorded before the freeze runs — Task 2 test "ignores snapshots observed at or after the deadline".
- No team records yet (0–0 everywhere) must show "Not available"/"–", never 0 points or ranks — Task 1 tests for `scoreEntry` and `rankEntries`.

import { CHALLENGE } from "@/config/challenge";
import { normalizeName } from "@/lib/league/commands";
import type { Side, TeamId } from "@/lib/types";
import type { ChallengePick, PickMap } from "./types";

export type SaveError = "invalid_picks" | "incomplete_entry" | "invalid_name";
export type ParsedSave = { ok: true; displayName: string; picks: PickMap } | { ok: false; error: SaveError };

const SIDES: ReadonlySet<unknown> = new Set<Side>(["OVER", "UNDER"]);

/** Checks a save request `{ displayName, picks: [{ teamId, side }] }`: exactly one side for every team, then the name. */
export function parseChallengeSave(body: Readonly<Record<string, unknown>>, teamIds: ReadonlySet<TeamId>): ParsedSave {
  if (!Array.isArray(body.picks)) return { ok: false, error: "invalid_picks" };
  const picks: Record<TeamId, Side> = {};
  for (const item of body.picks as unknown[]) {
    if (!item || typeof item !== "object") return { ok: false, error: "invalid_picks" };
    const { teamId, side } = item as { teamId?: unknown; side?: unknown };
    if (typeof teamId !== "string" || !teamIds.has(teamId) || !SIDES.has(side) || Object.hasOwn(picks, teamId)) {
      return { ok: false, error: "invalid_picks" };
    }
    picks[teamId] = side as Side;
  }
  if (Object.keys(picks).length !== teamIds.size) return { ok: false, error: "incomplete_entry" };
  const displayName = normalizeName(body.displayName, CHALLENGE.nameMaxLength);
  if (!displayName) return { ok: false, error: "invalid_name" };
  return { ok: true, displayName, picks };
}

/**
 * The picks to store for a save. A side that matches the stored one keeps its reference line and time; a new or
 * changed side takes the current line (null when there is none) and `now`. Sorted by team id.
 */
export function rebasePicks(
  stored: readonly ChallengePick[],
  submitted: PickMap,
  currentLines: Readonly<Record<TeamId, number>>,
  now: string,
): ChallengePick[] {
  const previous = new Map(stored.map((pick) => [pick.teamId, pick]));
  return Object.keys(submitted)
    .sort()
    .map((teamId) => {
      const side = submitted[teamId];
      const kept = previous.get(teamId);
      if (kept && kept.side === side) return kept;
      const line = currentLines[teamId];
      return { teamId, side, lineAtPick: Number.isFinite(line) ? line : null, pickedAt: now };
    });
}

export function pickMap(picks: readonly ChallengePick[]): PickMap {
  return Object.fromEntries(picks.map((pick) => [pick.teamId, pick.side]));
}

import { CHALLENGE } from "@/config/challenge";
import type { TeamId } from "@/lib/types";
import { pickMap } from "./picks";
import type { ChallengeEntry, ChallengePick, PickMap } from "./types";

/** The current line minus the line when the pick was saved. Null when either is missing or they are equal. */
export function lineMove(reference: number | null, current: number | null): number | null {
  if (reference === null || current === null) return null;
  const move = current - reference;
  return move === 0 ? null : move;
}

/** Saved picks whose working side is unchanged and whose current line differs from the pick's reference line. */
export function movedTeams(
  saved: readonly ChallengePick[],
  working: PickMap,
  current: Readonly<Record<TeamId, number>>,
): TeamId[] {
  return saved
    .filter((pick) => working[pick.teamId] === pick.side)
    .filter((pick) => lineMove(pick.lineAtPick, Number.isFinite(current[pick.teamId]) ? current[pick.teamId] : null) !== null)
    .map((pick) => pick.teamId);
}

export const SAVE_PICKS = "Save Picks";
export const SAVE_AGAINST_UPDATED = "Save Picks Against Updated Lines";

/** Informational only: saving never changes which lines score. */
export function saveLabel(movedCount: number): string {
  return movedCount > 0 ? SAVE_AGAINST_UPDATED : SAVE_PICKS;
}

/** True when the working picks or name differ from the saved entry (with nothing saved: anything picked or named). */
export function hasUnsavedChanges(saved: ChallengeEntry | null, working: PickMap, name: string): boolean {
  const savedPicks: PickMap = saved ? pickMap(saved.picks) : {};
  const teams = new Set([...Object.keys(savedPicks), ...Object.keys(working)]);
  for (const teamId of teams) if (savedPicks[teamId] !== working[teamId]) return true;
  const typed = normalizeName(name, CHALLENGE.nameMaxLength) ?? name.trim();
  return typed !== (saved?.displayName ?? "");
}

import type { Division } from "@/data/teams";
import type { Side, TeamInfo } from "@/lib/types";

export type TeamFilter = "all" | "unpicked" | "East" | "West";
export type TeamSort = "conference" | "team" | "line_desc" | "line_asc" | "movement";

export interface GridTeam {
  info: TeamInfo;
  division: Division;
  /** FanDuel's current line before the lock, the frozen line after; null when there is none. */
  line: number | null;
  /** Movement since the saved pick (see lineMove); null when none. */
  move: number | null;
  /** The working side; null while unpicked. */
  side: Side | null;
}

export const FILTER_OPTIONS: ReadonlyArray<{ value: TeamFilter; label: string }> = [
  { value: "all", label: "All Teams" },
  { value: "unpicked", label: "Unpicked" },
  { value: "East", label: "Eastern Conference" },
  { value: "West", label: "Western Conference" },
];

export const SORT_OPTIONS: ReadonlyArray<{ value: TeamSort; label: string }> = [
  { value: "conference", label: "Conference (A → Z)" },
  { value: "team", label: "Team (A → Z)" },
  { value: "line_desc", label: "Line (high → low)" },
  { value: "line_asc", label: "Line (low → high)" },
  { value: "movement", label: "Line movement (largest)" },
];

/** One of the filters, and with `movedOnly` only teams whose saved pick's line moved. */
export function filterTeams(teams: readonly GridTeam[], filter: TeamFilter, movedOnly: boolean): GridTeam[] {
  return teams.filter(
    (team) =>
      (filter === "all" || (filter === "unpicked" ? team.side === null : team.info.conference === filter)) &&
      (!movedOnly || team.move !== null),
  );
}

const fullName = (team: GridTeam) => `${team.info.city} ${team.info.name}`;
const byConference = (a: GridTeam, b: GridTeam) =>
  a.info.conference.localeCompare(b.info.conference) || fullName(a).localeCompare(fullName(b));
/** Nulls last; descending when `desc`. */
function compareNumbers(a: number | null, b: number | null, desc: boolean): number {
  if (a === null || b === null) return a === b ? 0 : a === null ? 1 : -1;
  return desc ? b - a : a - b;
}
const size = (move: number | null) => (move === null ? null : Math.abs(move));

export function sortTeams(teams: readonly GridTeam[], sort: TeamSort): GridTeam[] {
  const copy = [...teams];
  switch (sort) {
    case "team":
      return copy.sort((a, b) => fullName(a).localeCompare(fullName(b)));
    case "line_desc":
      return copy.sort((a, b) => compareNumbers(a.line, b.line, true) || byConference(a, b));
    case "line_asc":
      return copy.sort((a, b) => compareNumbers(a.line, b.line, false) || byConference(a, b));
    case "movement":
      return copy.sort((a, b) => compareNumbers(size(a.move), size(b.move), true) || byConference(a, b));
    default:
      return copy.sort(byConference);
  }
}

```ts
import { CHALLENGE } from "@/config/challenge";
import type { GameLog } from "@/lib/game-log/types";
import type { TeamId } from "@/lib/types";
import type { OpeningGame } from "./types";

/**
 * The season's first regular-season game across team game logs: the earliest dated game (logs already drop postponed
 * games and the NBA Cup Championship). Null when no game has a date. Pass every team's log: the opening game is in
 * both teams' logs, so one missing log can't hide it, but the server still requires all 30.
 */
export function openingGame(logs: readonly GameLog[]): OpeningGame | null {
  let best: { time: number; game: OpeningGame } | null = null;
  for (const log of logs) {
    for (const game of log.games) {
      if (game.date === null) continue;
      const time = Date.parse(game.date);
      if (Number.isNaN(time) || (best && best.time <= time)) continue;
      const teams: [TeamId | null, TeamId | null] =
        game.home === true ? [game.opponentId, log.teamId] : [log.teamId, game.opponentId];
      best = { time, game: { tipoff: new Date(time).toISOString(), teams } };
    }
  }
  return best?.game ?? null;
}

/** The lock time: the opening tip-off minus CHALLENGE.lockMinutesBeforeTipoff. */
export function challengeDeadline(tipoff: string, minutesBefore: number = CHALLENGE.lockMinutesBeforeTipoff): string {
  return new Date(Date.parse(tipoff) - minutesBefore * 60_000).toISOString();
}

import { isValidLine } from "@/lib/lines";
import type { TeamId } from "@/lib/types";
import type { LineSnapshot } from "./types";

export interface FreezeResult {
  values: Record<TeamId, number>;
  observedAt: Record<TeamId, string>;
  /** Sorted. */
  missing: TeamId[];
  /** The newest pre-deadline snapshot's source; null without one. */
  source: string | null;
}

/**
 * The shared scoring lines: for each team, its line in the newest snapshot observed strictly before the deadline that
 * has one. A team never observed before the deadline is missing — never filled in, never zero.
 */
export function freezeFromSnapshots(
  snapshots: readonly LineSnapshot[],
  deadline: string,
  teamIds: readonly TeamId[],
): FreezeResult {
  const cutoff = Date.parse(deadline);
  const before = snapshots
    .filter((snapshot) => Date.parse(snapshot.observedAt) < cutoff)
    .sort((a, b) => Date.parse(b.observedAt) - Date.parse(a.observedAt));
  const values: Record<TeamId, number> = {};
  const observedAt: Record<TeamId, string> = {};
  const missing: TeamId[] = [];
  for (const teamId of teamIds) {
    const snapshot = before.find((candidate) => isValidLine(candidate.lines[teamId]));
    if (snapshot) {
      values[teamId] = snapshot.lines[teamId];
      observedAt[teamId] = snapshot.observedAt;
    } else missing.push(teamId);
  }
  return { values, observedAt, missing: missing.sort(), source: before[0]?.source ?? null };
}
```

import type { TeamRecord } from "@/lib/records/types";
import { gamesPlayed, isSettled } from "@/lib/scoring";
import type { ChallengePhase, ChallengeSeason } from "./types";

export function challengePhase(
  season: Pick<ChallengeSeason, "deadline" | "frozen"> | null,
  now: Date,
  records: readonly TeamRecord[],
): ChallengePhase {
  if (season?.frozen) {
    if (records.length > 0 && records.every((record) => isSettled(record))) return "final";
    return records.some((record) => gamesPlayed(record) > 0) ? "in_season" : "locked";
  }
  if (!season?.deadline) return "schedule_unknown";
  return now.getTime() < Date.parse(season.deadline) ? "open" : "locking";
}

/** Entries are read-only in every phase but open and schedule_unknown. */
export function isLocked(phase: ChallengePhase): boolean {
  return phase !== "open" && phase !== "schedule_unknown";
}

```ts
import { CHALLENGE } from "@/config/challenge";
import { SCORING, type ScoringConfig } from "@/config/scoring";
import { evaluateCall, type Basis, type CallEvaluation } from "@/lib/scoring";
import type { TeamId, TeamInfo } from "@/lib/types";
import type { ChallengePick } from "./types";

export interface ScoredPick {
  pick: ChallengePick;
  team: TeamInfo;
  /** The frozen line; null when the team has none (unavailable for every entry). */
  line: number | null;
  /** Null when the line is missing. */
  projected: CallEvaluation | null;
  final: CallEvaluation | null;
}

export interface EntryScore {
  /** In the order of `picks`. */
  picks: ScoredPick[];
  /** Sum of projected points over picks whose team has played. Null until one has. */
  projectedTotal: number | null;
  projectedScored: number;
  /** Projected picks on the right side of the line (a push is not on pace). */
  onPace: number;
  /** Sum of final points. Null until every pick with a frozen line is settled. */
  finalTotal: number | null;
  finalScored: number;
  /** Picks without a frozen line. */
  unavailable: number;
}

const sum = (values: number[]) => values.reduce((total, value) => total + value, 0);

/** An entry against the shared frozen lines, through evaluateCall on both bases. */
export function scoreEntry(
  picks: readonly ChallengePick[],
  teams: readonly TeamInfo[],
  frozen: Readonly<Record<TeamId, number>>,
  config: ScoringConfig = SCORING,
): EntryScore {
  const byId = new Map(teams.map((team) => [team.id, team]));
  const scored = picks.map((pick): ScoredPick => {
    const team = byId.get(pick.teamId);
    if (!team) throw new Error(`Unknown team ${pick.teamId}`);
    const line = Number.isFinite(frozen[pick.teamId]) ? frozen[pick.teamId] : null;
    if (line === null) return { pick, team, line, projected: null, final: null };
    const withLine = { ...team, line };
    return {
      pick,
      team,
      line,
      projected: evaluateCall(pick.side, withLine, "projected", config),
      final: evaluateCall(pick.side, withLine, "final", config),
    };
  });
  const available = scored.filter((entry) => entry.line !== null);
  const projected = available.flatMap((entry) => (entry.projected?.status === "scored" ? [entry.projected] : []));
  const final = available.flatMap((entry) => (entry.final?.status === "scored" ? [entry.final] : []));
  return {
    picks: scored,
    projectedTotal: projected.length > 0 ? sum(projected.map((evaluation) => evaluation.points ?? 0)) : null,
    projectedScored: projected.length,
    onPace: projected.filter((evaluation) => evaluation.correct).length,
    finalTotal:
      available.length > 0 && final.length === available.length ? sum(final.map((evaluation) => evaluation.points ?? 0)) : null,
    finalScored: final.length,
    unavailable: scored.length - available.length,
  };
}

export function entryTotal(score: EntryScore, basis: Basis): number | null {
  return basis === "final" ? score.finalTotal : score.projectedTotal;
}

export interface RankedRow<T> {
  entry: T;
  score: number | null;
  /** Shared by tied scores (1, 1, 3). Null when unscored. */
  rank: number | null;
}

/**
 * Highest score first; ties (within CHALLENGE.tieTolerance) share a rank and list by name, then id. Unscored entries
 * come last, by name, without a rank — before any game every entry is unscored.
 */
export function rankEntries<T extends { id: string; displayName: string }>(
  rows: ReadonlyArray<{ entry: T; score: number | null }>,
  tolerance: number = CHALLENGE.tieTolerance,
): RankedRow<T>[] {
  const byName = (a: T, b: T) =>
    a.displayName.localeCompare(b.displayName, "en", { sensitivity: "base" }) || (a.id < b.id ? -1 : a.id > b.id ? 1 : 0);
  const scored = rows.filter((row): row is { entry: T; score: number } => row.score !== null);
  scored.sort((a, b) => (Math.abs(a.score - b.score) <= tolerance ? byName(a.entry, b.entry) : b.score - a.score));
  const ranked: RankedRow<T>[] = [];
  scored.forEach((row, index) => {
    const previous = ranked[index - 1];
    const tied = previous?.score != null && Math.abs(previous.score - row.score) <= tolerance;
    ranked.push({ entry: row.entry, score: row.score, rank: tied ? previous.rank : index + 1 });
  });
  const unscored = rows
    .filter((row) => row.score === null)
    .sort((a, b) => byName(a.entry, b.entry))
    .map((row) => ({ entry: row.entry, score: null, rank: null }));
  return [...ranked, ...unscored];
}

const iso = (date: Date | null) => date?.toISOString() ?? null;

function toSeason(row: SeasonRow): ChallengeSeason {
  return {
    season: row.season,
    openingGame: row.openingTipoff
      ? { tipoff: row.openingTipoff.toISOString(), teams: row.openingTeams ?? [null, null] }
      : null,
    deadline: iso(row.deadline),
    scheduleCheckedAt: iso(row.scheduleCheckedAt),
    scheduleError: row.scheduleError,
    frozen:
      row.lockedAt && row.frozenLines && row.frozenLineTimes && row.frozenMissing && row.frozenSource
        ? {
            values: row.frozenLines,
            observedAt: row.frozenLineTimes,
            missing: row.frozenMissing,
            source: row.frozenSource,
            lockedAt: row.lockedAt.toISOString(),
          }
        : null,
  };
}

export async function loadChallengeSeason(db: Db | Tx, season: number): Promise<ChallengeSeason | null> {
  const [row] = await db.select().from(challengeSeasons).where(eq(challengeSeasons.season, season));
  return row ? toSeason(row) : null;
}

/** The database clock now (not the transaction's start), for deadline checks made after waiting on a lock. */
export async function databaseNow(tx: Db | Tx): Promise<Date> {
  const [row] = await tx.select({ now: sql<Date>`clock_timestamp()`.mapWith(challengeSeasons.lockedAt) }).from(sql`(select 1) as clock`);
  return row.now;
}

export type ScheduleRead = { ok: true; game: OpeningGame } | { ok: false; error: string };

/**
 * Stores a schedule read for the season (creating its row): the opening game and deadline on success, only the error
 * on failure. Once the season is locked or its stored deadline has passed, nothing changes any more.
 */
export async function syncChallengeSchedule(db: Db, season: number, read: ScheduleRead, now: Date): Promise<ChallengeSeason> {
  return db.transaction(async (tx) => {
    await tx.insert(challengeSeasons).values({ season }).onConflictDoNothing();
    const [row] = await tx.select().from(challengeSeasons).where(eq(challengeSeasons.season, season)).for("update");
    if (row.lockedAt || (row.deadline && row.deadline.getTime() <= now.getTime())) return toSeason(row);
    const set = read.ok
      ? {
          openingTipoff: new Date(read.game.tipoff),
          openingTeams: read.game.teams,
          deadline: new Date(challengeDeadline(read.game.tipoff)),
          scheduleCheckedAt: now,
          scheduleError: null,
        }
      : { scheduleCheckedAt: now, scheduleError: read.error };
    const [updated] = await tx.update(challengeSeasons).set(set).where(eq(challengeSeasons.season, season)).returning();
    return toSeason(updated);
  });
}

function toSnapshot(row: typeof challengeLineSnapshots.$inferSelect): LineSnapshot {
  return { observedAt: row.observedAt.toISOString(), source: row.source, lines: row.lines };
}

export async function loadLatestSnapshot(db: Db, season: number): Promise<LineSnapshot | null> {
  const [row] = await db
    .select()
    .from(challengeLineSnapshots)
    .where(eq(challengeLineSnapshots.season, season))
    .orderBy(desc(challengeLineSnapshots.observedAt))
    .limit(1);
  return row ? toSnapshot(row) : null;
}

/**
 * Remembers a successful FanDuel read. Values identical to the newest snapshot only move its confirmed time; new
 * values add a snapshot observed at the read's time. The same read recorded twice is stored once.
 */
export async function recordLineSnapshot(db: Db, season: number, lines: LineSet): Promise<void> {
  const observedAt = new Date(lines.asOf);
  const [latest] = await db
    .select()
    .from(challengeLineSnapshots)
    .where(eq(challengeLineSnapshots.season, season))
    .orderBy(desc(challengeLineSnapshots.observedAt))
    .limit(1);
  if (latest && sameLineValues(latest.lines, lines.values)) {
    if (latest.confirmedAt.getTime() < observedAt.getTime()) {
      await db.update(challengeLineSnapshots).set({ confirmedAt: observedAt }).where(eq(challengeLineSnapshots.id, latest.id));
    }
    return;
  }
  await db
    .insert(challengeLineSnapshots)
    .values({ season, source: lines.source, observedAt, confirmedAt: observedAt, lines: { ...lines.values } })
    .onConflictDoNothing();
}

/**
 * Stores the shared frozen lines once the deadline has passed (database clock unless `now` is given). Idempotent:
 * the row lock serializes concurrent calls and a stored freeze is returned as it is. Null for an unknown season.
 */
export async function freezeChallenge(db: Db, season: number, options: { now?: Date } = {}): Promise<ChallengeSeason | null> {
  return db.transaction(async (tx) => {
    const [row] = await tx.select().from(challengeSeasons).where(eq(challengeSeasons.season, season)).for("update");
    if (!row) return null;
    if (row.lockedAt || !row.deadline) return toSeason(row);
    const now = options.now ?? (await databaseNow(tx));
    if (now.getTime() < row.deadline.getTime()) return toSeason(row);
    const snapshots = await tx
      .select()
      .from(challengeLineSnapshots)
      .where(and(eq(challengeLineSnapshots.season, season), lt(challengeLineSnapshots.observedAt, row.deadline)));
    const freeze = freezeFromSnapshots(snapshots.map(toSnapshot), row.deadline.toISOString(), TEAM_INFO.map((team) => team.id));
    const [updated] = await tx
      .update(challengeSeasons)
      .set({
        lockedAt: now,
        frozenLines: freeze.values,
        frozenLineTimes: freeze.observedAt,
        frozenMissing: freeze.missing,
        frozenSource: freeze.source ?? LINES.book,
      })
      .where(eq(challengeSeasons.season, season))
      .returning();
    return toSeason(updated);
  });
}

const ENTRY_ID_BYTES = 9; // 12 base64url characters

function createEntryId(): string {
  return base64url(crypto.getRandomValues(new Uint8Array(ENTRY_ID_BYTES)));
}

type EntryRow = typeof challengeEntries.$inferSelect;
type PickRow = typeof challengePicks.$inferSelect;

function toPick(row: PickRow): ChallengePick {
  return { teamId: row.teamId, side: row.side, lineAtPick: row.lineAtPick, pickedAt: row.pickedAt.toISOString() };
}

function toEntry(row: EntryRow, picks: PickRow[]): ChallengeEntry {
  return {
    id: row.id,
    season: row.season,
    displayName: row.displayName,
    createdAt: row.createdAt.toISOString(),
    savedAt: row.savedAt.toISOString(),
    picks: picks
      .filter((pick) => pick.entryId === row.id)
      .map(toPick)
      .sort((a, b) => (a.teamId < b.teamId ? -1 : a.teamId > b.teamId ? 1 : 0)),

async function withPicks(db: Db | Tx, rows: EntryRow[]): Promise<ChallengeEntry[]> {
  if (rows.length === 0) return [];
  const picks = await db.select().from(challengePicks).where(inArray(challengePicks.entryId, rows.map((row) => row.id)));
  return rows.map((row) => toEntry(row, picks));
}

export async function loadEntry(db: Db | Tx, entryId: string): Promise<ChallengeEntry | null> {

  return (await withPicks(db, rows))[0] ?? null;
}

export interface SaveEntryInput {
  season: number;
  sessionId: string;
  displayName: string;
  /** All 30 teams (checked by parseChallengeSave). */
  picks: PickMap;
  /** The server's current FanDuel lines: the reference for new or changed sides. */
  currentLines: Readonly<Record<TeamId, number>>;
}

class Abort extends Error {
  constructor(readonly code: DomainError) {
    super(code);
  }
}

/**
 * Saves the session's entry for the season in one transaction. The season row is locked FOR SHARE (a schedule change or
 * the freeze waits, and waits for this), then the clock is read: at or after the deadline, or once frozen, nothing is
 * written (`challenge_locked`); without a deadline, `deadline_unavailable`. Unchanged sides keep their reference line
 * and time (rebasePicks). `options.now` replaces the database clock in tests.
 */
export async function saveChallengeEntry(db: Db, input: SaveEntryInput, options: { now?: Date } = {}): Promise<Result<ChallengeEntry>> {
  try {
    return await db.transaction(async (tx) => {
      const [season] = await tx.select().from(challengeSeasons).where(eq(challengeSeasons.season, input.season)).for("share");
      if (!season?.deadline) throw new Abort("deadline_unavailable");
      const now = options.now ?? (await databaseNow(tx));
      if (season.lockedAt || now.getTime() >= season.deadline.getTime()) throw new Abort("challenge_locked");

      await tx
        .insert(challengeEntries)
        .values({ id: createEntryId(), season: input.season, ownerSessionId: input.sessionId, displayName: input.displayName, createdAt: now, savedAt: now })
        .onConflictDoNothing();
      const [entry] = await tx
        .select()
        .from(challengeEntries)
        .where(and(eq(challengeEntries.season, input.season), eq(challengeEntries.ownerSessionId, input.sessionId)))
        .for("update");
      if (!entry) throw new Error("Challenge entry was not created");
      const stored = await tx.select().from(challengePicks).where(eq(challengePicks.entryId, entry.id));
      const next = rebasePicks(stored.map(toPick), input.picks, input.currentLines, now.toISOString());
      await tx.delete(challengePicks).where(eq(challengePicks.entryId, entry.id));
      await tx.insert(challengePicks).values(
        next.map((pick) => ({ entryId: entry.id, teamId: pick.teamId, side: pick.side, lineAtPick: pick.lineAtPick, pickedAt: new Date(pick.pickedAt) })),
      );
      await tx.update(challengeEntries).set({ displayName: input.displayName, savedAt: now }).where(eq(challengeEntries.id, entry.id));
      const saved = await loadEntry(tx, entry.id);
      if (!saved) throw new Error("Challenge entry vanished");
      return succeed(saved);
    });
  } catch (error) {
    if (error instanceof Abort) return fail(error.code);
    throw error;
  }
}

import { timingSafeEqual } from "node:crypto";

/** Null when the request carries Vercel's `Bearer CRON_SECRET`; otherwise the response to send. */
export function cronRejection(request: Request): Response | null {
  const secret = process.env.CRON_SECRET;
  if (!secret) return Response.json({ error: "CRON_SECRET is not set." }, { status: 503 });
  const expected = Buffer.from(`Bearer ${secret}`);
  const given = Buffer.from(request.headers.get("authorization") ?? "");
  if (given.length === expected.length && timingSafeEqual(given, expected)) return null;
  return Response.json({ error: "unauthorized" }, { status: 401 });
}

import type { PickMap } from "@/lib/challenge/types";

export interface ChallengeDraft { picks: PickMap; name: string }
const key = (season: number) => `courtline.challenge.draft.v1.${season}`;
const listeners = new Set<() => void>();
let cache: { raw: string | null; value: ChallengeDraft | null } = { raw: null, value: null };

function read(season: number): string | null {
  try { return window.localStorage.getItem(key(season)); } catch { return null; }
}
export function subscribeDraft(listener: () => void): () => void {
  listeners.add(listener);
  window.addEventListener("storage", listener);
  return () => { listeners.delete(listener); window.removeEventListener("storage", listener); };
}
/** A stable object per stored string (useSyncExternalStore needs that). Null when absent or unreadable. */
export function getDraft(season: number): ChallengeDraft | null {
  const raw = read(season);
  if (raw === cache.raw) return cache.value;
  let value: ChallengeDraft | null = null;
  try {
    const parsed: unknown = raw ? JSON.parse(raw) : null;
    if (parsed && typeof parsed === "object" && "picks" in parsed && "name" in parsed) value = parsed as ChallengeDraft;
  } catch { value = null; }
  cache = { raw, value };
  return value;
}
export function writeDraft(season: number, draft: ChallengeDraft | null): void {
  try {
    if (draft) window.localStorage.setItem(key(season), JSON.stringify(draft));
    else window.localStorage.removeItem(key(season));
  } catch { /* storage unavailable: the page still works without a draft */ }
  listeners.forEach((listener) => listener());
}
```
